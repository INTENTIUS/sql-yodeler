# Rendered from .github/src/ci.ts by `npm run workflows`. Edit that file, not this one.

name: ci

on:
  pull_request: null
  push:
    branches:
      - main

permissions:
  contents: read

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: '${{ github.event_name == ''pull_request'' }}'

jobs:
  lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
      - run: npm ci
      - run: npx yodel --version
      - run: npm run typecheck
      - run: npm run lint
  test:
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        shard:
          - 1
          - 2
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
      - run: npm ci
      - run: npx vitest run --shard=${{ matrix.shard }}/2
  changes:
    runs-on: ubuntu-latest
    outputs:
      templates: '${{ steps.diff.outputs.templates }}'
    steps:
      - uses: actions/checkout@v4
      - id: diff
        env:
          EVENT: '${{ github.event_name }}'
          BASE: '${{ github.event.pull_request.base.sha }}'
        run: |
          if [ "$EVENT" != pull_request ]; then echo templates=true >> "$GITHUB_OUTPUT"; exit 0; fi
          git fetch -q --depth=1 origin "$BASE"
          if git diff --name-only "$BASE" HEAD | grep -qE '^(templates/|scripts/templates-check\.ts$|package\.json$|package-lock\.json$|\.github/src/(ci|templates)\.ts$)'; then
            echo templates=true >> "$GITHUB_OUTPUT"
          else
            echo "templates=false: the pull request changes nothing the templates check reads" | tee -a "$GITHUB_STEP_SUMMARY"
            echo templates=false >> "$GITHUB_OUTPUT"
          fi
  templates:
    needs:
      - changes
    if: needs.changes.outputs.templates == 'true'
    runs-on: ubuntu-latest
    timeout-minutes: 15
    strategy:
      fail-fast: false
      matrix:
        template:
          - clickhouse
          - postgres
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
      - run: npm ci
      - name: The template works with the published yodel
        run: npm run templates:check -- ${{ matrix.template }}
  check:
    needs:
      - lint
      - test
      - changes
      - templates
    if: always()
    runs-on: ubuntu-latest
    steps:
      - env:
          LINT: '${{ needs.lint.result }}'
          TEST: '${{ needs.test.result }}'
          CHANGES: '${{ needs.changes.result }}'
          TEMPLATES: '${{ needs.templates.result }}'
        run: |
          echo "lint: $LINT, test: $TEST, changes: $CHANGES, templates: $TEMPLATES"
          [ "$LINT" = success ] && [ "$TEST" = success ] && [ "$CHANGES" = success ] && { [ "$TEMPLATES" = success ] || [ "$TEMPLATES" = skipped ]; }
  docs-site:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v4
        with:
          lfs: true
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: docs-site/package-lock.json
      - name: npm ci
        working-directory: docs-site
        run: npm ci
      - name: Build the site
        working-directory: docs-site
        run: npm run build
  image:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
      - run: npm ci --ignore-scripts
      - name: Build the image from npm pack of this checkout
        run: |
          version=$(node -p 'require("./package.json").version')
          ctx=$(mktemp -d)
          npm pack --silent --pack-destination "$ctx"
          CHANT_PINS=$(node -p 'const l=require("./package-lock.json");Object.entries(l.packages).filter(([k,v])=>/^node_modules\/@intentius\/chant[^/]*$/.test(k)&&!v.dev).map(([k,v])=>k.slice(13)+"@"+v.version).join(" ")')
          docker build -f .github/image/Dockerfile --build-arg "YODEL_VERSION=$version" \
            --build-arg "YODEL_PACKAGE=/ctx/intentius-sql-yodeler-$version.tgz" --build-arg "CHANT_PINS=$CHANT_PINS" -t yodel-ci "$ctx"
      - name: yodel --version from the image
        run: docker run --rm yodel-ci yodel --version
      - name: The ClickHouse example builds and lints in the image, with no npm ci
        run: |
          cp -R examples/clickhouse "$RUNNER_TEMP/project"
          rm -rf "$RUNNER_TEMP/project/node_modules"
          docker run --rm -v "$RUNNER_TEMP/project:/work" -w /work yodel-ci sh -c 'chant build src --lexicon sql -o /tmp/schema.json && yodel lint'
